\section{Kiểm thử phần mềm}
\subsection{Các loại kiểm thử được sử dụng}
\label{subsec:test-types}

Để bảo đảm chất lượng toàn diện cho Hệ thống Quản lý Nhân sự Crewzy, nhóm phát triển đã áp dụng một chiến lược kiểm thử đa tầng, kết hợp ba loại kiểm thử: \textbf{Unit testing} (kiểm thử đơn vị), \textbf{API testing} và \textbf{Integration testing} (kiểm thử tích hợp).

\subsubsection*{Kiểm thử đơn vị (Unit testing)}
\textbf{Mục đích:} Xác minh tính đúng đắn của từng đơn vị mã nhỏ nhất (hàm, phương thức, component React) một cách độc lập, không phụ thuộc vào các thành phần ngoài như cơ sở dữ liệu thật hay dịch vụ mạng.  
\textbf{Phạm vi:} Toàn bộ các hàm xử lý nghiệp vụ trong backend, các component giao diện, context, service và route handler.  
\textbf{Công cụ:}
\begin{itemize}
    \item \textbf{Frontend:} \texttt{Jest} (test runner mặc định từ Create React App) kết hợp với \texttt{React Testing Library} để kiểm thử từ góc nhìn người dùng (render, fireEvent, waitFor).
    \item \textbf{Backend:} \texttt{Vitest} (framework tương thích Jest, tốc độ cao) cùng với \texttt{Supertest} để mô phỏng HTTP request đến Express app; sử dụng \texttt{vi.mock()} để mô phỏng pg pool.
\end{itemize}
\textbf{Chiến lược thực hiện:}
\begin{itemize}
    \item Với frontend: tập trung kiểm tra render ban đầu, tương tác người dùng, thay đổi trạng thái, hiển thị có điều kiện (loading, lỗi), phân quyền giao diện (ProtectedRoute) và các thao tác bất đồng bộ (gọi API, mock service).
    \item Với backend: kiểm tra độc lập middleware xác thực (trả 401/403), kiểm tra route handler với dữ liệu đầu vào không hợp lệ (400) và hợp lệ (200/201), mock database để kiểm thử truy vấn SQL và biến đổi dữ liệu mà không cần kết nối thật.
\end{itemize}
Nhờ đó, nhóm có thể phản hồi nhanh, chạy toàn bộ bộ test trong vài giây và đạt độ phủ cao mà không gặp rủi ro từ môi trường ngoài.

\subsubsection*{Kiểm thử API (API testing)}
\textbf{Mục đích:} Xác nhận các endpoint REST hoạt động đúng theo đặc tả, kiểm tra định dạng phản hồi (JSON schema), mã trạng thái HTTP, xác thực và phân quyền ở tầng HTTP, độc lập với giao diện.  
\textbf{Phạm vi:} Tất cả các API công khai thuộc 8 module: \texttt{auth}, \texttt{employees}, \texttt{leaves}, \texttt{expenses}, \texttt{kpi}, \texttt{attendance}, \texttt{salary}, \texttt{health}.  
\textbf{Công cụ:}
\begin{itemize}
    \item \texttt{Postman} (xây dựng bộ sưu tập \texttt{postman\_collection.json}) gồm 49 request, sử dụng kịch bản tiền điều kiện (Pre‑request Script) để tạo dữ liệu duy nhất và trích xuất token, kịch bản kiểm thử (Test Script) để kiểm tra khẳng định.
    \item \texttt{Newman} (CLI runner) cho phép chạy bộ sưu tập trên môi trường CI hoặc local, xuất kết quả ra \texttt{test\_run.json}.
    \item \texttt{Python parser} (tùy chọn) để chuyển đổi kết quả Newman thành báo cáo tùy chỉnh.
\end{itemize}
\textbf{Chiến lược thực hiện:}
\begin{itemize}
    \item Chuỗi xác thực: Đăng nhập thành công $\rightarrow$ Lưu \texttt{adminToken}, \texttt{adminUserId} vào biến môi trường $\rightarrow$ Gán tự động vào header \texttt{Authorization} cho các request tiếp theo.
    \item Mỗi kịch bản kiểm thử bao gồm cả trường hợp thành công (happy path) lẫn thất bại (thiếu token, sai vai trò, dữ liệu không hợp lệ).
    \item API testing được chạy trực tiếp trên cơ sở dữ liệu Railway MySQL để phản ánh đúng môi trường production, do đó cũng kiêm vai trò integration test nhẹ.
\end{itemize}

\subsubsection*{Kiểm thử tích hợp (Integration testing)}
\textbf{Mục đích:} Đánh giá sự phối hợp giữa nhiều thành phần trong hệ thống (route + middleware + database thật) theo các luồng nghiệp vụ phức tạp, vượt ra ngoài phạm vi của một đơn vị đơn lẻ.  
\textbf{Phạm vi:} Các luồng chính như: đăng nhập $\rightarrow$ tạo đơn nghỉ phép $\rightarrow$ phê duyệt bởi quản lý; tạo yêu cầu chi tiêu $\rightarrow$ phê duyệt/từ chối; cập nhật KPI $\rightarrow$ tính tổng kết; chấm công check‑in/check‑out.  
\textbf{Công cụ:}
\begin{itemize}
    \item Các test case tích hợp sử dụng \texttt{Supertest} với database thật (Railway MySQL), không mock pg pool, nhằm xác nhận câu truy vấn, ràng buộc toàn vẹn và kết quả trả về chính xác.
    \item Các test case Postman/Newman cho các API cũng đồng thời là kiểm thử tích hợp vì gọi đến server đang chạy với CSDL thật.
\end{itemize}
\textbf{Chiến lược thực hiện:}
\begin{itemize}
    \item Kết hợp Unit test cho các luồng xử lý nghiệp vụ cốt lõi với dữ liệu mock và Integration test cho các trường hợp yêu cầu tương tác thật với CSDL.
    \item Mỗi module được tích hợp riêng, sau đó kiểm thử dây chuyền giữa các module để bảo đảm tính nhất quán toàn hệ thống.
\end{itemize}

\subsection{Unit testing}
\label{sec:unit-testing}

Kiểm thử đơn vị là nền tảng của chiến lược đảm bảo chất lượng, giúp phát hiện sớm lỗi trong từng thành phần riêng lẻ của hệ thống. Phần này mô tả chi tiết cách tiếp cận, công cụ và kết quả kiểm thử đơn vị cho cả frontend (giao diện React) và backend (máy chủ Node.js/Express).

\subsubsection{Frontend testing (Jest)}
\label{sec:frontend-unit}

\textbf{Môi trường thực thi:}
\begin{itemize}
    \item \textbf{Test Runner:} Jest (phiên bản 27.5.1, được tích hợp sẵn từ Create React App).
    \item \textbf{Thư viện hỗ trợ:} React Testing Library (phiên bản 12.1.5) – ưu tiên kiểm thử từ góc nhìn người dùng thông qua các truy vấn như \texttt{getByRole}, \texttt{getByLabelText}, \texttt{findByText}.
    \item \textbf{Mock dữ liệu:} \texttt{jest.mock()} để giả lập các API service bên ngoài; \texttt{MemoryRouter} để cung cấp ngữ cảnh định tuyến.
    \item \textbf{Tốc độ:} Tất cả 75 test case chạy trong 17,4\,s.
\end{itemize}

\textbf{Chiến lược kiểm thử:}
\begin{enumerate}
    \item \textbf{Render cơ bản:} Xác minh giao diện khởi tạo hiển thị đúng các phần tử (input, nút bấm, nhãn). Ví dụ: form đăng nhập có ô nhập email và mật khẩu.
    \item \textbf{Tương tác người dùng:} Kích hoạt sự kiện (click, thay đổi input, chọn option) và kiểm tra phản ứng của component (gọi hàm xử lý, cập nhật state).
    \item \textbf{Hiển thị có điều kiện:} Mô phỏng các trạng thái như đang tải (loading), lỗi từ API (sai mật khẩu), hoặc danh sách rỗng. Sử dụng \texttt{waitFor} để chờ cập nhật bất đồng bộ.
    \item \textbf{Phân trang và lọc dữ liệu:} Kiểm tra chức năng tìm kiếm, lọc theo từ khoá, số lượng dòng hiển thị sau phân trang.
    \item \textbf{Bảo vệ định tuyến (ProtectedRoute):} Đảm bảo người dùng chưa xác thực bị chuyển hướng đến trang đăng nhập, và người dùng không đủ quyền bị chặn truy cập.
    \item \textbf{Context và dịch vụ:} Kiểm thử AuthContext cung cấp đúng giá trị xác thực; ApiService gửi đúng HTTP method và trả về dữ liệu đã mock.
\end{enumerate}

\textbf{Mẫu thiết kế test case tiêu biểu:}
\begin{itemize}
    \item \texttt{test('renders Login form with email and password inputs')} – kiểm tra render tĩnh.
    \item \texttt{test('submits form with valid credentials')} – kiểm tra tương tác và gọi hàm.
    \item \texttt{test('displays error message when login fails')} – mock API thất bại và chờ hiển thị thông báo lỗi.
    \item \texttt{test('filters employee list by search term')} – nhập từ khóa, xác nhận số dòng còn lại.
    \item \texttt{test('ProtectedRoute redirects unauthorized users to login')} – kiểm tra phân quyền với context giả lập.
\end{itemize}

\subsubsection{Backend testing (Vitest)}
\label{sec:backend-unit}

\textbf{Môi trường thực thi:}
\begin{itemize}
    \item \textbf{Test Runner:} Vitest 4.1.4 – tương thích cú pháp Jest, cho tốc độ vượt trội khi chạy song song.
    \item \textbf{HTTP assertion:} Supertest 6.x – gửi yêu cầu đến ứng dụng Express mà không cần khởi động server thật.
    \item \textbf{Giả lập CSDL:} \texttt{vi.mock()} thay thế pool của \texttt{pg} bằng các stub trả về dữ liệu cứng để kiểm tra luồng xử lý mà không phụ thuộc vào hạ tầng thật.
    \item \textbf{Coverage:} Tích hợp \texttt{vitest coverage} xuất báo cáo dạng JSON và HTML.
\end{itemize}

\textbf{Chiến lược kiểm thử:}
\begin{enumerate}
    \item \textbf{Kiểm thử middleware xác thực:} Gọi trực tiếp hàm middleware với request giả lập (thiếu header, token sai định dạng, token hết hạn) để khẳng định mã trạng thái 401/403 và nội dung thông báo.
    \item \textbf{Kiểm thử route handler:} Gửi request qua Supertest với payload không hợp lệ (email sai, thiếu trường) để kiểm tra validation 400; với dữ liệu hợp lệ để nhận 200/201.
    \item \textbf{Tích hợp CSDL giả lập:} Mock \texttt{pool.query} trả về các dòng cụ thể, sau đó gọi API và kiểm tra cấu trúc JSON trả về, bao gồm các trường như \texttt{emp\_id}, \texttt{leave\_id}, \texttt{status}, v.v.
    \item \textbf{Kiểm thử phân quyền:} Gán token của một nhân viên thông thường, thử truy cập dữ liệu của nhân viên khác hoặc các endpoint chỉ dành cho admin để đảm bảo nhận được 403.
    \item \textbf{Happy path:} Chuỗi tạo mới – ví dụ tạo đơn nghỉ phép thành công, kiểm tra mã 201 và sự tồn tại của \texttt{leave\_id} trong phản hồi.
\end{enumerate}

\textbf{Các tệp kiểm thử chính:}
\begin{itemize}
    \item \texttt{auth.middleware.unit.test.js} – 3 bài kiểm tra bao phủ 100\% middleware.
    \item \texttt{auth.routes.unit.test.js} + các tệp con – kiểm tra đăng nhập/đăng ký (thành công và thất bại).
    \item \texttt{employees.routes.db.unit.test.js} – kiểm tra lấy danh sách, chi tiết, tạo mới với DB mock.
    \item \texttt{leaves.routes.db.unit.test.js} – 4 bài kiểm tra cho luồng quản lý nghỉ phép.
    \item \texttt{salary.routes.db.unit.test.js} – 3 bài kiểm tra, đạt 87.5\% độ phủ tuyến đường.
    \item Các tệp “unit” khác chỉ kiểm tra rào cản xác thực (auth gating) cho các module còn lại.
\end{itemize}

\subsection{API và Integration testing}
\label{sec:api-integration}

Kiểm thử API và tích hợp tập trung vào việc xác nhận các điểm cuối REST hoạt động chính xác trên môi trường gần với thực tế nhất – sử dụng cơ sở dữ liệu Railway MySQL thật. Phần này trình bày chiến lược triển khai và phân tích kết quả đạt được.

\subsubsection{Chiến lược kiểm thử}
\label{subsec:api-strategy}

\textbf{Công cụ và quy trình:}
\begin{itemize}
    \item \textbf{Postman Collections} – toàn bộ 49 request được tổ chức thành 8 module: \texttt{auth}, \texttt{employees}, \texttt{leaves}, \texttt{expenses}, \texttt{kpi}, \texttt{attendance}, \texttt{salary}, \texttt{health}. Mỗi request bao gồm phương thức HTTP, URL, header, body (nếu POST/PUT) và hai kịch bản:
    \begin{itemize}
        \item \textit{Pre‑request Script} – dùng để tạo dữ liệu ngẫu nhiên (email duy nhất, ngày tháng động) và thiết lập biến môi trường.
        \item \textit{Test Script} – viết các khẳng định (assertions) về mã trạng thái, cấu trúc JSON, thời gian phản hồi và trích xuất token cho các request sau.
    \end{itemize}
    \item \textbf{Newman} (CLI runner) – thực thi bộ sưu tập trên môi trường CI/local với câu lệnh \texttt{npm run test:api:run}, xuất kết quả chi tiết ra tệp \texttt{test\_run.json} để phân tích.
    \item \textbf{Môi trường thực thi} – toàn bộ request được gửi đến máy chủ Express đang chạy thật, kết nối trực tiếp đến cơ sở dữ liệu Railway MySQL, do đó đồng thời đóng vai trò kiểm thử tích hợp đầu cuối.
\end{itemize}

\textbf{Chuỗi xác thực (Authentication Chain):}
Để tự động hoá luồng có xác thực, các biến môi trường sau được sử dụng: \texttt{adminToken}, \texttt{adminUserId}, \texttt{testEmail}, \texttt{testPassword}. Chuỗi hoạt động như sau:
\begin{enumerate}
    \item \texttt{POST /api/auth/login} – gửi thông tin đăng nhập admin, nhận token và \texttt{emp\_id}.
    \item Test Script trích xuất \texttt{response.token} và \texttt{response.user.emp\_id}, gán vào biến môi trường.
    \item Tất cả các request tiếp theo tự động gắn header \texttt{Authorization: Bearer \{\{adminToken\}\}}.
\end{enumerate}
Nhờ đó, mỗi lần chạy lại bộ sưu tập sẽ tự động đăng nhập và duy trì phiên làm việc mà không cần can thiệp thủ công.

\textbf{Kịch bản kiểm thử tiêu biểu:}
\begin{itemize}
    \item \textbf{Xác thực thành công:} \texttt{POST /auth/login} và \texttt{POST /auth/register} với dữ liệu hợp lệ $\rightarrow$ kỳ vọng 200/201, token tồn tại.
    \item \textbf{Xác thực thất bại:} thiếu email, sai mật khẩu, thiếu trường bắt buộc $\rightarrow$ 400 hoặc 401 với thông báo tiếng Việt.
    \item \textbf{Truy cập trái phép:} gọi endpoint mà không có token $\rightarrow$ 401.
    \item \textbf{Phân quyền:} dùng token của nhân viên để truy cập API chỉ dành cho admin $\rightarrow$ 403 (``Không có quyền truy cập'').
    \item \textbf{CRUD có xác thực:} tạo mới, đọc danh sách, cập nhật, xóa các đối tượng (nghỉ phép, chi tiêu, KPI, chấm công) với token hợp lệ $\rightarrow$ kiểm tra mã 200/201 và các trường dữ liệu quan trọng.
    \item \textbf{Toàn vẹn dữ liệu:} mỗi phản hồi thành công được kiểm tra có đúng định dạng JSON, chứa các trường bắt buộc như \texttt{emp\_id}, \texttt{leave\_id}, \texttt{status}, v.v.
    \item \textbf{Hiệu năng cơ bản:} xác nhận thời gian phản hồi trung bình dưới 500\,ms cho từng request.
\end{itemize}

\textbf{Tổ chức bộ sưu tập:}
Bộ sưu tập được chia nhỏ thành từng fragment theo module, sau đó hợp nhất vào một tệp \texttt{postman\_collection.json}. Cấu trúc thư mục như sau:
\begin{itemize}
    \item \textbf{Auth} – 6 request (login, register với các biến thể thành công/ lỗi).
    \item \textbf{Employees} – 8 request (CRUD, profile, danh sách nghỉ phép/lương của nhân viên).
    \item \textbf{Leaves} – 8 request (quản lý đơn nghỉ phép, phê duyệt, từ chối).
    \item \textbf{Expenses} – 8 request (tạo, duyệt, từ chối chi tiêu, báo cáo theo tháng).
    \item \textbf{KPI} – 8 request (quản lý chỉ số KPI, tổng kết theo kỳ).
    \item \textbf{Attendance} – 7 request (chấm công check‑in, check‑out, lịch sử, báo cáo).
    \item \textbf{Salary} – 3 request (xem bảng lương, cập nhật/tính toán).
    \item \textbf{Health} – 2 request (kiểm tra dịch vụ và kết nối DB).
\end{itemize}
